cmake_minimum_required(VERSION 3.25)
project(weave_consumer LANGUAGES CXX)
if(MSVC)
  string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
  set(CMAKE_VS_GLOBALS "VcpkgEnabled=false;VcpkgApplocalDeps=false")
endif()

find_package(weave CONFIG REQUIRED COMPONENTS ${WEAVE_COMPONENT})
if(WEAVE_FIND_ONLY)
  return()
endif()
# Repeated lookup and optional missing components must also be supported.
find_package(weave CONFIG REQUIRED COMPONENTS core OPTIONAL_COMPONENTS absent)
if(weave_absent_FOUND)
  message(FATAL_ERROR "Unknown optional component was reported as available")
endif()
foreach(component IN ITEMS core io runtime tcp)
  if(component IN_LIST WEAVE_EXPECTED_MODULES)
    if(NOT TARGET weave::${component})
      message(FATAL_ERROR "Missing transitive target weave::${component}")
    endif()
    get_target_property(dependencies weave::${component} INTERFACE_LINK_LIBRARIES)
    if(dependencies)
      string(REGEX MATCHALL "weave::[a-z]+" dependencies "${dependencies}")
      foreach(dependency IN LISTS dependencies)
        string(REPLACE "weave::" "" dependency "${dependency}")
        if(NOT dependency IN_LIST WEAVE_EXPECTED_MODULES)
          message(FATAL_ERROR "Unwanted link dependency: ${dependency}")
        endif()
      endforeach()
    endif()
  elseif(TARGET weave::${component})
    message(FATAL_ERROR "Unrequested component target leaked into the consumer: ${component}")
  endif()
  if(NOT component IN_LIST WEAVE_INSTALLED_MODULES AND EXISTS "${WEAVE_PREFIX}/include/weave/${component}.hpp")
    message(FATAL_ERROR "Unselected component headers were installed: ${component}")
  endif()
endforeach()

enable_testing()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE weave::${WEAVE_COMPONENT})
string(TOUPPER "${WEAVE_COMPONENT}" component_define)
target_compile_definitions(consumer PRIVATE WEAVE_USE_${component_define}=1)
add_test(NAME installed_consumer COMMAND consumer)
set_tests_properties(installed_consumer PROPERTIES TIMEOUT 15)

# Every installed header must compile on its own, without native OS headers.
file(GLOB_RECURSE headers RELATIVE "${WEAVE_PREFIX}/include" "${WEAVE_PREFIX}/include/*.hpp")
set(probes "")
foreach(header IN LISTS headers)
  string(MAKE_C_IDENTIFIER "${header}" id)
  set(probe "${CMAKE_CURRENT_BINARY_DIR}/headers/${id}.cpp")
  file(WRITE "${probe}" "#include <${header}>\n#if defined(_WINDOWS_) || defined(_WINSOCKAPI_) || defined(_WINSOCK2API_)\n#error Public header leaked Windows headers\n#endif\n")
  list(APPEND probes "${probe}")
endforeach()
add_library(header_checks OBJECT ${probes})
target_link_libraries(header_checks PRIVATE weave::${WEAVE_COMPONENT})
