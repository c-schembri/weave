foreach(example IN ITEMS query prepared stream pool blocking primitives batch tls)
  add_executable(mysql_${example} ${example}.cpp)
  target_link_libraries(mysql_${example} PRIVATE weave::mysql)
  weave_no_exceptions(mysql_${example})
  if(MSVC)
    target_compile_definitions(mysql_${example} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  weave_tls_runtime(mysql_${example})
endforeach()

add_executable(mysql_migrations migrations/main.cpp)
target_link_libraries(mysql_migrations PRIVATE weave::mysql)
weave_no_exceptions(mysql_migrations)
if(MSVC)
  target_compile_definitions(mysql_migrations PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
weave_tls_runtime(mysql_migrations)
weave_add_migrations(TARGET mysql_migrations DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/migrations/changes" NAME mysql_migrations)
