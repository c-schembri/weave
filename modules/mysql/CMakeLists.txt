find_package(OpenSSL 3.5 REQUIRED COMPONENTS Crypto)
add_library(weave_mysql src/connection.cpp src/row.cpp src/pool.cpp src/blocking.cpp
  src/migration_compiler.cpp src/migrations.cpp)
weave_module(mysql)
target_link_libraries(weave_mysql PUBLIC weave::database weave::tls PRIVATE OpenSSL::Crypto)
target_link_libraries(weave_mysql PRIVATE $<BUILD_INTERFACE:weave_database_async>)

if(WEAVE_BUILD_TESTS)
  add_test(NAME mysql_benchmark_tooling COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_tooling.py")
  set_tests_properties(mysql_benchmark_tooling PROPERTIES LABELS tooling)
  weave_add_test(weave_mysql_tests tests/mysql.cpp)
  target_link_libraries(weave_mysql_tests PRIVATE weave::mysql OpenSSL::Crypto)
  target_include_directories(weave_mysql_tests PRIVATE "${PROJECT_SOURCE_DIR}/test_support")
  weave_tls_runtime(weave_mysql_tests)
  weave_add_test(weave_mysql_migration_tests tests/migrations.cpp)
  target_link_libraries(weave_mysql_migration_tests PRIVATE weave::mysql)
  target_include_directories(weave_mysql_migration_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
  weave_tls_runtime(weave_mysql_migration_tests)
  if(TARGET weave_runtime)
    target_link_libraries(weave_mysql_tests PRIVATE weave::runtime)
    target_compile_definitions(weave_mysql_tests PRIVATE WEAVE_MYSQL_TEST_RUNTIME=1)
  endif()

  add_executable(weave_mysql_live_tests tests/live.cpp)
  target_link_libraries(weave_mysql_live_tests PRIVATE weave::mysql)
  weave_tls_runtime(weave_mysql_live_tests)
  if(TARGET weave_runtime)
    target_link_libraries(weave_mysql_live_tests PRIVATE weave::runtime)
    target_compile_definitions(weave_mysql_live_tests PRIVATE WEAVE_MYSQL_TEST_RUNTIME=1)
  endif()
  set(WEAVE_MYSQL_TEST_SERVER "" CACHE FILEPATH "Optional mysqld for isolated live qualification")
  set(WEAVE_MYSQL_TEST_CLIENT "" CACHE FILEPATH "Matching mysql executable for fixture setup")
  if((WEAVE_MYSQL_TEST_SERVER AND NOT WEAVE_MYSQL_TEST_CLIENT) OR
      (WEAVE_MYSQL_TEST_CLIENT AND NOT WEAVE_MYSQL_TEST_SERVER))
    message(FATAL_ERROR "MySQL live tests require both WEAVE_MYSQL_TEST_SERVER and WEAVE_MYSQL_TEST_CLIENT")
  endif()
  if(WEAVE_MYSQL_TEST_SERVER AND WEAVE_MYSQL_TEST_CLIENT)
    add_test(NAME mysql_live COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/live.py"
      --server "${WEAVE_MYSQL_TEST_SERVER}" --client "${WEAVE_MYSQL_TEST_CLIENT}"
      --test "$<TARGET_FILE:weave_mysql_live_tests>")
    set_tests_properties(mysql_live PROPERTIES TIMEOUT 90 LABELS integration)
    if(WEAVE_ENABLE_ASAN AND MSVC)
      set_tests_properties(mysql_live PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${WEAVE_COMPILER_BIN}")
    endif()
  endif()
endif()
