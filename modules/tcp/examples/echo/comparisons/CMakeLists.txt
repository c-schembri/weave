include(FetchContent)
block()
  # Keep upstream's optional TLS, DNS and test dependencies out of this TCP example.
  set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
  set(CMAKE_POLICY_DEFAULT_CMP0126 NEW)
  set(BUILD_DOC OFF)
  set(BUILD_C-ARES OFF)
  set(BUILD_TESTING OFF)
  set(BUILD_SHARED_LIBS OFF)
  set(USE_SPDLOG OFF)
  set(TRANTOR_USE_TLS none)
  FetchContent_Declare(trantor GIT_REPOSITORY https://github.com/an-tao/trantor.git
    GIT_TAG 63a4e5e164e219dc3bf30cdbfa1462ae5602fa97 GIT_SUBMODULES "")
  FetchContent_MakeAvailable(trantor)
endblock()
if(MSVC)
  # Follow upstream's exception model without changing Weave's compilation policy.
  target_compile_options(trantor PRIVATE /EHsc)
endif()
if(WEAVE_ENABLE_ASAN AND MSVC)
  target_compile_options(trantor PRIVATE /fsanitize=address /Zi)
endif()

foreach(library IN ITEMS weave asio libuv usockets trantor)
  add_executable(echo_${library} ${library}.cpp)
  set_target_properties(echo_${library} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/examples/echo_servers/$<CONFIG>")
  target_compile_features(echo_${library} PRIVATE cxx_std_23)
  target_compile_definitions(echo_${library} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
  if(library STREQUAL "trantor")
    if(MSVC)
      target_compile_options(echo_trantor PRIVATE /W4 /permissive- /Zc:preprocessor /EHsc)
    else()
      target_compile_options(echo_trantor PRIVATE -Wall -Wextra -Wpedantic -fexceptions)
    endif()
  else()
    weave_no_exceptions(echo_${library})
  endif()
  if(WEAVE_ENABLE_ASAN AND MSVC)
    target_compile_options(echo_${library} PRIVATE /fsanitize=address /Zi)
    target_link_options(echo_${library} PRIVATE /INCREMENTAL:NO /DEBUG)
  endif()
  if(WEAVE_BUILD_TESTS)
    add_test(NAME echo_${library}_integration
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/echo_example.py"
        --executable "$<TARGET_FILE:echo_${library}>")
    set_tests_properties(echo_${library}_integration PROPERTIES TIMEOUT 45 LABELS examples)
    if(WEAVE_ENABLE_ASAN AND MSVC)
      set_tests_properties(echo_${library}_integration PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${WEAVE_COMPILER_BIN}")
    endif()
  endif()
endforeach()

target_link_libraries(echo_weave PRIVATE weave::tcp weave::runtime)
target_include_directories(echo_asio SYSTEM PRIVATE ${asio_SOURCE_DIR}/asio/include)
target_compile_definitions(echo_asio PRIVATE ASIO_STANDALONE ASIO_NO_EXCEPTIONS)
target_link_libraries(echo_asio PRIVATE weave::core ws2_32 mswsock)
target_link_libraries(echo_libuv PRIVATE weave::core uv_a)
target_link_libraries(echo_usockets PRIVATE weave::core weave_usockets)

# The standard-only parser needs headers, not core's propagated no-exceptions flags.
target_include_directories(echo_trantor PRIVATE "$<TARGET_PROPERTY:weave::core,INTERFACE_INCLUDE_DIRECTORIES>")
target_link_libraries(echo_trantor PRIVATE trantor)
