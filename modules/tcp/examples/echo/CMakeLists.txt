foreach(execution IN ITEMS context runtime)
  if(execution STREQUAL "runtime" AND NOT TARGET weave_runtime)
    continue()
  endif()
  foreach(style IN ITEMS minimal stream)
    if(style STREQUAL "minimal")
      set(example echo_server_${execution}_minimal)
    else()
      set(example echo_server_stream_${execution})
    endif()
    add_executable(${example} ${style}/${execution}.cpp)
    target_link_libraries(${example} PRIVATE weave::tcp)
    if(execution STREQUAL "runtime")
      target_link_libraries(${example} PRIVATE weave::runtime)
    endif()
    weave_no_exceptions(${example})
    set_target_properties(${example} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
    if(WEAVE_BUILD_TESTS)
      add_test(NAME ${example}_integration
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/echo_example.py"
          --executable "$<TARGET_FILE:${example}>" --usage-exit-code 1)
      set_tests_properties(${example}_integration PROPERTIES TIMEOUT 45 LABELS examples)
      if(WEAVE_ENABLE_ASAN AND MSVC)
        set_tests_properties(${example}_integration PROPERTIES
          ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${WEAVE_COMPILER_BIN}")
      endif()
    endif()
  endforeach()
endforeach()
if(WEAVE_BUILD_BENCHMARKS AND TARGET weave_runtime)
  # The concurrent Weave comparison uses Runtime; native comparison dependencies stay private.
  add_subdirectory(comparisons)
endif()
