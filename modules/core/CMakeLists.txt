add_library(weave_core INTERFACE)
weave_module(core)
target_compile_features(weave_core INTERFACE cxx_std_23)
target_compile_definitions(weave_core INTERFACE WEAVE_RECYCLE_FRAMES=$<BOOL:${WEAVE_RECYCLE_FRAMES}>)
if(MSVC)
  target_compile_options(weave_core INTERFACE /permissive- /Zc:preprocessor /EHs-c-)
  target_compile_definitions(weave_core INTERFACE _HAS_EXCEPTIONS=0)
else()
  target_compile_options(weave_core INTERFACE -fno-exceptions)
endif()
if(WEAVE_PROFILE_RUNTIME)
  target_compile_definitions(weave_core INTERFACE WEAVE_PROFILE_RUNTIME=1)
endif()
if(WEAVE_ENABLE_ASAN)
  if(MSVC)
    target_compile_options(weave_core INTERFACE /fsanitize=address /Zi)
    target_link_options(weave_core INTERFACE /INCREMENTAL:NO /DEBUG)
  else()
    target_compile_options(weave_core INTERFACE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(weave_core INTERFACE -fsanitize=address)
  endif()
endif()
if(WEAVE_BUILD_TESTS)
  find_package(Threads REQUIRED)
  weave_add_test(weave_core_tests tests/task.cpp tests/frame_allocator.cpp tests/log.cpp)
  target_link_libraries(weave_core_tests PRIVATE weave::core Threads::Threads)
endif()
