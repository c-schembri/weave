foreach(example IN ITEMS columns mapping)
  add_executable(database_${example} adapters/${example}.cpp)
  target_link_libraries(database_${example} PRIVATE weave::database)
  weave_no_exceptions(database_${example})
  set_target_properties(database_${example} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
  if(WEAVE_BUILD_TESTS)
    add_test(NAME database_${example}_example COMMAND database_${example})
    set_tests_properties(database_${example}_example PROPERTIES TIMEOUT 15 LABELS examples)
    if(WEAVE_ENABLE_ASAN AND MSVC)
      set_tests_properties(database_${example}_example PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${WEAVE_COMPILER_BIN}")
    endif()
  endif()
endforeach()
